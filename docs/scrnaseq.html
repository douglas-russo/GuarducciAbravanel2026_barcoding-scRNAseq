2  scRNA-seq analyses

2.1 Preliminaries

Raw and processed scRNAseq files are deposited on GEO under accession number GSE326012. Raw fastqs are used as inputs to nf-core/scrnaseq v2.7.1. We include as processed data R object files (rds) of the following three types:

  1. Each individual sample’s unfiltered cells. Filename format: scrnaseq-seurat-object-unfiltered-.rds

  2. Each individual sample’s QC-metric-filtered cells. Filename format: scrnaseq-seurat-object-qcfiltered-processed-.rds. Contains predicted cell cycle scores, phase, GLM-PCA embeddings, UMAP-GLM-PCA embeddings, and kmeans clustering assignments in object slots for each sample.

  3. All individual sample’s QC-metric-filtered cells merged together into a single object. Filename: scrnaseq-seurat-object-qcfiltered-processed-all-samples.rds. Contains predicted cell cycle scores, phase, GLM-PCA embeddings, UMAP-GLM-PCA embeddings, and kmeans clustering on all cells in object slots. The ‘orig.ident’ column of the ‘meta.data’ object slot identifies the sample of origin of a cell and matches the “library name” sample-level metadata field. Dimensionality reductions and clustering were done on all merged cells together.

We recommend using pre-computed files from (2) and (3) above for downstream analyses to ensure consistency with paper figures. In cases where preprocessing steps may be computationally intensive or introduce non-deterministic outputs, we load in pre-computed files.

#' Collect outputs of nf-core scrnaseq alevin-fry via simpleaf quant (sq), load
#' into a gene x cell matrix, then collapse gene IDs to
#' gene symbols (summing), following Jeremy Simon's tutorial:
#'
#' https://jeremymsimon.github.io/HSPH_scRNA_20241023/HSPH_scRNA_20241023.html#import-alevin-fry-counts-into-r-with-fishpondloadfry
#'
#' Note, this keeps all cells without removing any emptyDroplets cells. This
#' corresponds to individual processed data files per sample of the naming
#' scheme
#'
#' "<GEO-sample-accession-number>_scrnaseq-seurat-object-unfiltered-<sample-name>.rds"
#'
#' deposited in GEO.
#'
#' @param sample_name One of c(Parental, WT_Vehicle, Mut_Vehicle, WT_PalboR,
#' WT_PalboR_Clone, Mut_PalboR, WT_AbemaR, Mut_AbemaR)
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns Seurat object of unfiltered counts.
create_count_mtx_from_nfcore_scrnaseq_unfiltered <- function(
  sample_name,
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  idir = sprintf(
    "/rafalab/drusso/rjlab/cdk46i/data/processed/12-nfcore-scrnaseq/alevin/%s_alevin_results/af_quant",
    sample_name
  )

  sample_rna <- loadFry(idir, outputFormat = "snRNA")

  tx2gene <- read.table(
    "/rafalab/drusso/rjlab/cdk46i/data/processed/12-nfcore-scrnaseq/alevin/salmon/ref/gene_id_to_name.tsv",
    header = FALSE,
    sep = "\t",
    col.names = c("tx", "gene")
  )

  # Use matrix multiplication to group and sum counts by gene symbol
  exp.txId <- rownames(counts(sample_rna))
  exp.geneId <- as.vector(tx2gene$gene[match(exp.txId, tx2gene$tx)])
  exp.tx.grp <- t(sparse.model.matrix(~ 0 + exp.geneId))

  sample_rna.summarized <- exp.tx.grp %*% counts(sample_rna)
  rownames(sample_rna.summarized) <- rownames(sample_rna.summarized) |>
    str_replace_all(".+.geneId", "")

  out <- CreateSeuratObject(counts = sample_rna.summarized)
  out$orig.ident <- sample_name # for easy merging of all samples later on into a single Seurat object

  saveRDS(
    out,
    file.path(
      ddir,
      sprintf("scrnaseq-seurat-object-unfiltered-%s.rds", sample_name)
    )
  )
  return(out)
}

2.3.3 Load in function to read in nf-core/scrnaseq outputs into Seurat objects.

#' Collect outputs of nf-core scrnaseq alevin-fry via simpleaf quant (sq), load
#' into a gene x cell matrix, remove emptyDroplets, then collapse gene IDs to
#' gene symbols (summing), following Jeremy Simon's tutorial:
#'
#' https://jeremymsimon.github.io/HSPH_scRNA_20241023/HSPH_scRNA_20241023.html#import-alevin-fry-counts-into-r-with-fishpondloadfry
#'
#' Note, paths will need to be changed to match the location of the
#' nf-core/scrnaseq run outputs.
#'
#' @param sample_name One of c(Parental, WT_Vehicle, Mut_Vehicle, WT_PalboR,
#' WT_PalboR_Clone, Mut_PalboR, WT_AbemaR, Mut_AbemaR)
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns Seurat object of emptyDrops-filtered counts.
create_count_mtx_from_nfcore_scrnaseq <- function(
  sample_name,
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  # Change prefix of path to alevin output directory of nf-core/scrnaseq output
  idir = sprintf(
    "/rafalab/drusso/rjlab/cdk46i/data/processed/12-nfcore-scrnaseq/alevin/%s_alevin_results/af_quant",
    sample_name
  )

  sample_rna <- loadFry(idir, outputFormat = "snRNA")

  e.out <- DropletUtils::emptyDrops(counts(sample_rna))
  is.cell <- e.out$FDR <= 0.01
  is.cell[is.na(is.cell)] = FALSE
  table(is.cell)

  # Remove empty droplets
  gex_q.filt <- sample_rna[, is.cell]

  tx2gene <- read.table(
    "/rafalab/drusso/rjlab/cdk46i/data/processed/12-nfcore-scrnaseq/alevin/salmon/ref/gene_id_to_name.tsv",
    header = FALSE,
    sep = "\t",
    col.names = c("tx", "gene")
  )

  # Use matrix multiplication to group and sum counts by gene symbol
  exp.txId <- rownames(counts(gex_q.filt))
  exp.geneId <- as.vector(tx2gene$gene[match(exp.txId, tx2gene$tx)])
  exp.tx.grp <- t(sparse.model.matrix(~ 0 + exp.geneId))

  gex_q.filt.summarized <- exp.tx.grp %*% counts(gex_q.filt)
  rownames(gex_q.filt.summarized) <- rownames(gex_q.filt.summarized) |>
    str_replace_all(".+.geneId", "")

  out <- CreateSeuratObject(counts = gex_q.filt.summarized)
  out$orig.ident <- sample_name # for easy merging of all samples later on into a single Seurat object

  saveRDS(
    e.out,
    file.path(ddir, sprintf("simpleaf-quant-emptyDrops-%s.rds", sample_name))
  )
  saveRDS(
    out,
    file.path(ddir, sprintf("simpleaf-quant-dds-%s.rds", sample_name))
  )
  return(out)
}

#' Merge Seurat objects of multiple samples, add in percent mitochondrial
#' genes.
#'
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns SeuratObject of merged samples.
merge_seurat_simpleaf <- function(
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  idir = "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-eda"
  Parental <- readRDS(file.path(idir, "simpleaf-quant-dds-Parental.rds"))
  WT_Vehicle <- readRDS(file.path(idir, "simpleaf-quant-dds-WT_Vehicle.rds"))
  Mut_Vehicle <- readRDS(file.path(idir, "simpleaf-quant-dds-Mut_Vehicle.rds"))
  WT_PalboR <- readRDS(file.path(idir, "simpleaf-quant-dds-WT_PalboR.rds"))
  WT_PalboR_Clone <- readRDS(file.path(
    idir,
    "simpleaf-quant-dds-WT_PalboR_Clone.rds"
  ))
  Mut_PalboR <- readRDS(file.path(idir, "simpleaf-quant-dds-Mut_PalboR.rds"))
  WT_AbemaR <- readRDS(file.path(idir, "simpleaf-quant-dds-WT_AbemaR.rds"))
  Mut_AbemaR <- readRDS(file.path(idir, "simpleaf-quant-dds-Mut_AbemaR.rds"))

  sce <- merge(
    Parental,
    y = c(
      WT_Vehicle,
      Mut_Vehicle,
      WT_PalboR,
      WT_PalboR_Clone,
      Mut_PalboR,
      WT_AbemaR,
      Mut_AbemaR
    ),
    add.cell.ids = c(
      "Parental",
      "WT_Vehicle",
      "Mut_Vehicle",
      "WT_PalboR",
      "WT_PalboR_Clone",
      "Mut_PalboR",
      "WT_AbemaR",
      "Mut_AbemaR"
    ),
    project = "cdk46i"
  )
  sce <- PercentageFeatureSet(
    sce,
    pattern = "^MT-",
    col.name = "percent.mt"
  )
  # Ordering for plot panels, legends, etc.
  sce$orig.ident <- factor(
    sce$orig.ident,
    levels = c(
      "Mut_AbemaR",
      "WT_AbemaR",
      "Mut_PalboR",
      "WT_PalboR",
      "WT_PalboR_Clone",
      "Mut_Vehicle",
      "WT_Vehicle",
      "Parental"
    )
  )

  saveRDS(sce, file.path(ddir, sprintf("simpleaf-quant-dds-allsamples.rds")))
  return(sce)
}

2.3.6 Load in function to add QC information to Seurat object of all samples.

#' Plot QC metrics of scRNAseq cells, filter dataset according to given fixed
#' thresholds.
#'
#' Uses fixed thresholds as opposed to something like miseq, as the distribution
#' of the QC metrics, makes the mixture model miseq uses a poor fit.
#'
#' @param thresh_max_percent.mt Maximum percent.mt for which cells are kept.
#' @param thresh_min_nFeature_RNA Min nFeature_RNA for which cells are kept.
#' @param thresh_min_nCount_RNA Min nCount_RNA for which cells are kept.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns Seurat object of filtered data.
filter_eda_qc <- function(
  thresh_max_percent.mt = 25,
  thresh_min_nFeature_RNA = 100,
  thresh_min_nCount_RNA = 1e4,
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  dds <- readRDS("data/processed/02-scrnaseq/simpleaf-quant-dds-allsamples.rds")

  # Filter the Seurat object, and write to file.
  dds <- dds[, dds$percent.mt <= thresh_max_percent.mt]
  dds <- dds[, dds$nFeature_RNA >= thresh_min_nFeature_RNA]
  dds <- dds[, dds$nCount_RNA >= thresh_min_nCount_RNA]

  dds <- JoinLayers(dds)
  dds <- dds[rowSums(dds@assays$RNA@layers$counts) > 0, ] # filter out genes with zero counts across all cells

  saveRDS(
    dds,
    file.path(ddir, sprintf("simpleaf-quant-dds-allsamples_filterqc.rds"))
  )
  return(dds)
}

#' Merge glmpca embeddings as a slot in a Seurat object of just those cells,
#' calculate cellcycle phases, umap on the glmpca imputs, and kmeans
#' clustering on the data.
#'
#' @param infile_seurat_rds Path to RDS of Seurat object.
#' @param infile_seurat_rds Path to rds of GLM-PCA output.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
merge_seurat_glmpca_umap_kmeans_simpleaf <- function(
  infile_seurat_rds = "data/processed/02-scrnaseq/simpleaf-quant-dds-allsamples_filterqc.rds",
  # Paths need to be changed to point to fastglmpca output files.
  infile_fastglmpca = c(
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_Mut_AbemaR.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_WT_AbemaR.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_Mut_PalboR.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_WT_PalboR.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_WT_PalboR_Clone.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_Mut_Vehicle.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_WT_Vehicle.rds",
    "/rafalab/drusso/rjlab/cdk46i/data/processed/07-sc-simpleaf-fastglmpca-kraken/fastglmpca-K50-scrytop2500bydev_simpleaf-quant-dds-allsamples_filterqc_Parental.rds"
  ),
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  dds_full <- readRDS(infile_seurat_rds)
  dds_full <- NormalizeData(dds_full) # only for calculating Seurat's cell cycle scoring, not for any other analyses.
  s.genes <- base::intersect(cc.genes$s.genes, rownames(dds_full))
  g2m.genes <- base::intersect(cc.genes$g2m.genes, rownames(dds_full))
  dds_full <- CellCycleScoring(
    dds_full,
    s.features = s.genes,
    g2m.features = g2m.genes,
    set.ident = FALSE
  )

  # Loop through GLM-PCA fits, process each separately.
  for (f in infile_fastglmpca) {
    prefix <- sprintf(
      ifelse(
        grepl("excludecc", f),
        "%s_excludecc_glmpca-umap-kmeans",
        "%s_glmpca-umap-kmeans"
      ),
      gsub(
        ".rds",
        "",
        gsub("^fastglmpca-K50.*scrytop2500bydev_", "", basename(f))
      )
    )

    outfile_rds <- file.path(ddir, sprintf("%s.rds", prefix))
    outfile_kmeans_output_rds <- file.path(
      ddir,
      sprintf("%s_kmeans-output.rds", gsub("_glmpca-umap-kmeans", "", prefix))
    )

    outfile_plt_inertia_pdf <- file.path(
      odir,
      sprintf("%s_kmeans-inertia.pdf", prefix)
    )
    outfile_plt_silhouette_pdf <- file.path(
      odir,
      sprintf("%s_kmeans-silhouette.pdf", prefix)
    )
    outfile_plt_inertia_png <- file.path(
      odir,
      sprintf("%s_kmeans-inertia.png", prefix)
    )
    outfile_plt_silhouette_png <- file.path(
      odir,
      sprintf("%s_kmeans-silhouette.png", prefix)
    )

    print(outfile_rds)
    fit <- readRDS(f)
    dds <- dds_full[, rownames(fit$V)] # only take cells in glm-pca fit from entire dataset.

    # Add the glmpca gene embeddings to the Seurat object
    dds[["fastglmpca"]] <- CreateDimReducObject(
      embeddings = fit$V,
      key = "k_",
      assay = DefaultAssay(dds)
    )
    set.seed(101341)

    set.seed(385)
    umap.fit <- uwot::umap(fit$V)
    dds[["umap_fastglmpca"]] <- CreateDimReducObject(
      embeddings = umap.fit,
      key = "",
      assay = DefaultAssay(dds)
    )

    # kmeans clustering
    dist.matrix <- dist(Embeddings(dds, reduction = "fastglmpca"))

    lst = purrr::map(2:10, \(k) {
      print(k)

      # kmeans cluster the glm-pca embeddings
      clst.kmeans <- kmeans(
        Embeddings(dds, reduction = "fastglmpca"),
        k,
        iter.max = 25,
        nstart = 25
      )
      dds@meta.data <<- dds@meta.data |> # global assignment
        mutate(!!sym(sprintf("kmeans%s", k)) := clst.kmeans$cluster)

      centers <- clst.kmeans$centers
      clusters <- clst.kmeans$cluster

      # Calculate inertia
      # sum of square distances of each point to its closest (assigned) cluster centroid
      inertia <- sum(
        (Embeddings(dds, reduction = "fastglmpca") - centers[clusters, ])^2
      )
      clst.kmeans$inertia <- inertia

      # Calculate average silhouette score
      sil <- cluster::silhouette(x = clst.kmeans$cluster, dist = dist.matrix)
      meansil <- mean(sil[, 3]) # average silhouette width over all cells.
      clst.kmeans$silhouette <- meansil

      return(clst.kmeans)
    })

    st = unique(dds$orig.ident)
    # Inertia and Silhouette scores for choosing a suitable k
    kmeans_metrics <- tibble(
      k = 2:10,
      inertia = purrr::map_dbl(lst, \(x) x$inertia),
      silhouette = purrr::map_dbl(lst, \(x) x$silhouette)
    )

    (plt_inertia <- kmeans_metrics |>
      ggplot(aes(x = k, y = inertia)) +
      geom_point() +
      labs(
        title = "Inertia of kmeans clusterings",
        subtitle = paste0(st, collapse = " "),
        x = "Number of kmeans clusters",
        y = "Inertia (sum of squares)"
      ) +
      scale_x_continuous(breaks = 2:10) +
      theme_bw())

    (plt_silhouette <- kmeans_metrics |>
      ggplot(aes(x = k, y = silhouette)) +
      geom_point() +
      labs(
        title = "Average silhouette scores of kmeans clusterings",
        subtitle = paste0(st, collapse = " "),
        x = "Number of kmeans clusters",
        y = "Average silhouette score"
      ) +
      scale_x_continuous(breaks = 2:10) +
      theme_bw())

    # Save plots of inertia and silhouette scores
    ggsave(outfile_plt_inertia_pdf, plt_inertia, width = 6, height = 6)
    ggsave(outfile_plt_silhouette_pdf, plt_silhouette, width = 6, height = 6)
    ggsave(
      outfile_plt_inertia_png,
      plt_inertia,
      width = 6,
      height = 6,
      units = "in"
    )
    ggsave(
      outfile_plt_silhouette_png,
      plt_silhouette,
      width = 6,
      height = 6,
      units = "in"
    )

    # Save seurat objects with kmeans clustering information.
    saveRDS(lst, outfile_kmeans_output_rds)
    saveRDS(dds, outfile_rds)
  }
}

#' Per sample, calculate matrices of cell-cell Euclidean distances on GLM-PCA
#' cell embeddings for 1000 sampled cells.
#'
#' @param infile_seurat_rds Path to rds of Seurat object of all samples.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
get_cell_cell_dists_within_sample_fastglmpca_embeddings_simpleaf <- function(
  infile_seurat_rds = "data/processed/02-scrnaseq/simpleaf-quant-dds-allsamples_filterqc_glmpca-umap-kmeans.rds",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"

  prefix <- gsub(".rds", "", basename(infile_seurat_rds))
  dds <- readRDS(infile_seurat_rds)
  Idents(dds) <- "orig.ident"

  gene.embedding.tbl <- Embeddings(dds, "fastglmpca") |>
    as.data.frame() |>
    tibble::rownames_to_column("barcode_id") |>
    tibble::as_tibble()

  gene.embedding.tbl.meta <- dds@meta.data |>
    tibble::rownames_to_column("barcode_id") |>
    inner_join(gene.embedding.tbl, by = "barcode_id") |>
    group_by(orig.ident)

    purrr::map(\(df) {

        tibble::column_to_rownames("barcode_id") |>

        tibble::rownames_to_column("barcode_id_1") |>

          !barcode_id_1,

    purrr::list_rbind() |>

2.5.1 Estimate genewise dispersions and expression rates per sample.

#' Calculate genewise dispersions (phi) per each of the 8 samples.
#'
#' These dispersions get used in calculating the Wald statistic of the difference
#' in average expression between groups when doing differential expression and
#' gene set testing.
#'
#' We allow each sample to have its own genewise dispersions, as these may
#' differ among the 8 samples.
#'
#' @param infile_seurat_rds Path to Seurat object rds.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns Data frame of overdispersion and rate (lambda) terms per gene, per sample.
get_genewise_rates_and_dispersions_per_sample <- function(
  infile_seurat_rds = "data/processed/02-scrnaseq/simpleaf-quant-dds-allsamples_filterqc_glmpca-umap-kmeans.rds",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  dds <- readRDS(infile_seurat_rds)
  cts <- SeuratObject::GetAssayData(dds, layer = "counts")
  prefix <- gsub(".rds", "", basename(infile_seurat_rds))

  phi_df <- c(
    "Mut_AbemaR",
    "WT_AbemaR",
    "Mut_PalboR",
    "WT_PalboR",
    "WT_PalboR_Clone",
    "Mut_Vehicle",
    "WT_Vehicle",
    "Parental"
  ) |>
    purrr::map_dfr(\(b) {
      cts <- SeuratObject::GetAssayData(
        dds[, dds$orig.ident == b],
        layer = "counts"
      ) # gene x cells orientation
      n <- Matrix::colSums(cts)
      N <- sum(n)
      lam <- Matrix::rowSums(cts) / N # gene-specific rates
      rates <- outer(lam, n)
      phi <- Matrix::rowSums((cts - rates)^2 / rates) / (ncol(cts) - 1) # estimated dispersion
      stat <- lam / sqrt(pmin(phi, 1) * lam)

      phi |>
        as.data.frame() |>
        cbind(lam) |>
        cbind(stat) |>
        tibble::rownames_to_column("gene") |>
        dplyr::mutate(N_reads = N) |>
        dplyr::mutate(N_cells = ncol(cts)) |>
        dplyr::mutate(orig.ident = b)
    })

  readr::write_csv(
    phi_df,
    file.path(ddir, sprintf("genewise-overdispersions_%s.csv", prefix))
  )
  saveRDS(
    phi_df,
    file.path(ddir, sprintf("genewise-overdispersions_%s.rds", prefix))
  )

  return(phi_df)
}

get_genewise_rates_per_cluster_within_sample <- function(

    purrr::map_dfr(\(x) {

        dds[,
          dds@meta.data |> pull(!!sym(sprintf("kmeans%i", cluster_var))) == x

  cts <- SeuratObject::GetAssayData(dds, layer = "counts")
  prefix <- gsub(".rds", "", basename(infile_seurat_rds))

  lam_df <- 1:cluster_var |>
    purrr::map(\(x) {
      purrr::map(c("G1", "G2M", "S"), \(phase) { # cell cycle phase within cluster
        print(sprintf(
          "%s:%s %i",
          x,
          phase,
          sum(
            ((dds@meta.data |> pull(!!sym(sprintf("kmeans%i", cluster_var))) ==
              x) &
              (dds$Phase == phase))
          )
        ))

        if (
          sum(
            (dds@meta.data |> pull(!!sym(sprintf("kmeans%i", cluster_var))) ==
              x) &
              (dds$Phase == phase)
          ) <
            min_cells
        ) {
          return(NULL) # too few cells to calculate rate confidently
        }

        cts <- SeuratObject::GetAssayData(
          dds[,
            ((dds@meta.data |> pull(!!sym(sprintf("kmeans%i", cluster_var))) ==
              x) &
              (dds$Phase == phase))
          ],
          layer = "counts"
        ) # gene x cells orientation
        n <- Matrix::colSums(cts)
        N <- sum(n)
        lam <- Matrix::rowSums(cts) / N # gene-specific rates

        lam |>
          as.data.frame() |>
          tibble::rownames_to_column("gene") |>
          dplyr::mutate(N_reads = N) |>
          dplyr::mutate(N_cells = ncol(cts)) |>
          dplyr::mutate(orig.ident = unique(dds$orig.ident)) |>
          dplyr::mutate(kmclust = sprintf("kmeans%i", cluster_var)) |>
          dplyr::mutate(cluster = x) |>
          dplyr::mutate(phase = phase) |>
          dplyr::mutate(cluster_phase = sprintf("%s:%s", cluster, phase))
      }) |>
        purrr::list_rbind()
    }) |>
    purrr::list_rbind() |>
    dplyr::filter(N_cells >= min_cells) # require at least min_cells cells to confidently call rate within a cluster

  readr::write_csv(
    lam_df,
    file.path(
      ddir,
      sprintf(
        "genewise-rates-within-kmeans%i-within-phase_geq%icells_%s.csv",
        cluster_var,
        min_cells,
        prefix
      )
    )
  )
  saveRDS(
    lam_df,
    file.path(
      ddir,
      sprintf(
        "genewise-rates-within_kmeans%i-within-phase_geq%icells_%s.rds",
        cluster_var,
        min_cells,
        prefix
      )
    )
  )
  return(lam_df)
}

2.5.5 Calculate quasi-Poisson Wald statistics comparing two samples at a time.

#' Calculate the Wald statistic in a quasiPoisson glm that compares two 
#' samples at a time per gene.
#'
#' Uses the sample-specific, per-gene estimated dispersions calculated above
#' as part of the standard error of the tested coefficient in the QP glm.
#'
#' @param infile_seurat_rds Path to rds of Seurat object.
#' @param infile_dispersions Path to rds of estimated genewise dispersion per 
#' sample.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns Data frame of Wald statistics comparing samples and sample 
#' specific expression rates and dispersion estimates.
quasipoisson_glm_de_simpleaf <- function(
  infile_seurat_rds = "data/processed/02-scrnaseq/simpleaf-quant-dds-allsamples_filterqc_glmpca-umap-kmeans.rds",
  infile_dispersions = "data/processed/02-scrnaseq/genewise-overdispersions_simpleaf-quant-dds-allsamples_filterqc_glmpca-umap-kmeans.rds",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  prefix <- gsub(".rds", "", basename(infile_seurat_rds))
  disps <- readRDS(infile_dispersions)

  disps_A = disps |> filter(orig.ident == "Mut_AbemaR")
  disps_B = disps |> filter(orig.ident == "WT_AbemaR")
  stopifnot(all(disps_A$gene == disps_B$gene))

  disps_wide <- disps |>
    pivot_wider(
      id_cols = gene,
      names_from = orig.ident,
      values_from = c("phi", "lam", "stat", "N_cells", "N_reads")
    )

  nms <- c(
    "Mut_AbemaR",
    "WT_AbemaR",
    "Mut_PalboR",
    "WT_PalboR",
    "WT_PalboR_Clone",
    "Mut_Vehicle",
    "WT_Vehicle",
    "Parental"
  )
  tidyr::expand_grid(x = nms, y = nms) |>
    dplyr::filter(x != y) |>
    purrr::pwalk(\(x, y) {
      disps_wide <<- disps_wide |> # global assignment
        dplyr::mutate(
          !!sym(sprintf("waldstat_%s_minus_%s", x, y)) := ((log(
            !!sym(sprintf("lam_%s", x))
          ) -
            log(!!sym(sprintf("lam_%s", y)))) /
            sqrt(
              (pmax(1, !!sym(sprintf("phi_%s", x))) /
                (!!sym(sprintf("lam_%s", x)) *
                  !!sym(sprintf("N_reads_%s", x)))) +
                (pmax(1, !!sym(sprintf("phi_%s", y))) /
                  (!!sym(sprintf("lam_%s", y)) *
                    !!sym(sprintf("N_reads_%s", y))))
            ))
        )
    })

  saveRDS(
    disps_wide,
    file.path(
      ddir,
      sprintf("waldstat-ranked-genes-across-samples_%s.rds", prefix)
    )
  )
  return(disps_wide)
}

2.5.8 Calculate quasi-Poisson Wald statistics comparing clusters within a single sample within a cell cycle phase.

#' Calculate the Wald statistic in a quasiPoisson glm that compares two 
#' clusters of cells within the same cell cycle phase within a single sample.
#'
#' Uses the sample-specific, per-gene estimated dispersions calculated above
#'  as part of the standard error of the tested coefficient in the QP glm.
#'
#' @param infile_rates Path to rds of estimated genewise rates per cluster 
#' within cell cycle phase for a single sample.
#' @param infile_dispersions Path to rds of estimated genewise dispersion per 
#' sample.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns Data frame of Wald statistics comparing clusters within each cell 
#' cycle phase, (cluster x cell cycle phase) specific expression rates, and 
#' sample specific dispersion estimates.
quasipoisson_glm_de_across_clusters_within_sample_per_phase_simpleaf <- function(
  infile_rates,
  infile_dispersions = "data/processed/02-scrnaseq/genewise-overdispersions_simpleaf-quant-dds-allsamples_filterqc_glmpca-umap-kmeans.rds",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  prefix <- gsub(".rds", "", basename(infile_rates))
  rates <- readRDS(infile_rates)
  k = as.numeric(gsub("^genewise-rates-within_kmeans([0-9]+).*", "\\1", prefix))

  disps <- readRDS(infile_dispersions) |>
    dplyr::filter(orig.ident == unique(rates$orig.ident))

  rates_wide <- rates |>
    tidyr::pivot_wider(
      id_cols = gene,
      names_from = cluster_phase,
      values_from = c("lam", "N_reads", "N_cells")
    ) |>
    dplyr::left_join(disps, by = "gene")

  purrr::map(c("G1", "G2M", "S"), \(phase) {
    tidyr::expand_grid(x = 1:k, y = 1:k) |>
      dplyr::filter(x != y) |>
      purrr::pwalk(\(x, y) {
        if (
          !(sprintf("lam_%s:%s", x, phase) %in% colnames(rates_wide)) |
            !(sprintf("lam_%s:%s", y, phase) %in% colnames(rates_wide))
        ) {
          # Not enough cells,
          print(sprintf(
            "Not enough cells for either %s:%s or %s:%s.",
            x,
            phase,
            y,
            phase
          ))
          return()
        } else {
          rates_wide <<- rates_wide |> # global assignment
            dplyr::mutate(
              !!sym(sprintf(
                "waldstat_cluster%s-%s_minus_cluster%s-%s",
                x,
                phase,
                y,
                phase
              )) := ((log(!!sym(sprintf("lam_%s:%s", x, phase))) -
                log(!!sym(sprintf("lam_%s:%s", y, phase)))) /
                sqrt(
                  (pmax(1, phi) /
                    (!!sym(sprintf("lam_%s:%s", x, phase)) *
                      !!sym(sprintf("N_reads_%s:%s", x, phase)))) +
                    (pmax(1, phi) /
                      (!!sym(sprintf("lam_%s:%s", y, phase)) *
                        !!sym(sprintf("N_reads_%s:%s", y, phase))))
                ))
            )
        }
      })
  })

  (plt_hist <- rates_wide |>
    dplyr::select(gene, starts_with("waldstat")) |>
    tidyr::pivot_longer(
      cols = !gene,
      names_to = "comparison",
      values_to = "waldstat"
    ) |>
    dplyr::mutate(
      comparison_title = gsub(
        "minus_",
        "minus\n",
        gsub("waldstat_", "", comparison)
      )
    ) |>
    ggplot(aes(x = pmax(pmin(waldstat, 10), -10))) +
    geom_histogram() +
    labs(title = sprintf("%s Wald statistics", unique(rates_wide$orig.ident))) +
    theme_bw() +
    facet_wrap(~comparison_title))
  ggplot2::ggsave(
    file.path(
      odir,
      sprintf(
        "waldstat-ranked-genes-across-cluster-within-sample-per-phase-histogram_%s.png",
        prefix
      )
    ),
    plt_hist,
    width = 8,
    height = 8
  )

  saveRDS(
    rates_wide,
    file.path(
      ddir,
      sprintf(
        "waldstat-ranked-genes-across-cluster_within-sample-per-phase_%s.rds",
        prefix
      )
    )
  )
  return(rates_wide)
}

2.5.9 Calculate cameraPR gene set testing across two samples within a cell cycle phase.

#' Run limma::cameraPR gene set testing for Hallmark gene sets for contrasts 
#' of interest within each cell cycle phase on gene lists ranked by 
#' quasi-Poisson Wald statistics.
#'
#' @param infile_waldstats Data frame of Wald statistics comparing samples 
#' within a cell cycle phase and (sample and cell cycle phase) specific 
#' expression rates and sample specific dispersion estimates.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
run_cameraPR_across_sample_within_phase_1v1_contrasts_quasipoisson_glm_simpleaf <- function(
  infile_waldstats = "data/processed/02-scrnaseq/waldstat-ranked-genes-across-sample_within-phase_genewise-rates-within-phase-per-sample_simpleaf-quant-dds-allsamples_filterqc_glmpca-umap-kmeans.rds",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  set.seed(1039101)
  prefix <- gsub(".rds", "", basename(infile_waldstats))

  ws <- readRDS(infile_waldstats)
  hk <- msigdbr::msigdbr(species = "human", category = "H") # Hallmark gene collection
  phk = split(x = hk$human_gene_symbol, f = hk$gs_name)

  (contrastdf <- rbind(
    c("Mut_AbemaR", "Mut_Vehicle"),
    c("WT_AbemaR", "WT_Vehicle"),
    c("Mut_PalboR", "Mut_Vehicle"),
    c("WT_PalboR", "WT_Vehicle"),

    c("Mut_Vehicle", "WT_Vehicle"),
    c("Mut_PalboR", "WT_PalboR"),
    c("Mut_AbemaR", "WT_AbemaR")
  ) |>
    as_tibble())
  names(contrastdf) <- c("x", "y")

  rbind(
    contrastdf |> dplyr::mutate(x = str_c(x, "_S"), y = str_c(y, "_S")),
    contrastdf |> dplyr::mutate(x = str_c(x, "_G1"), y = str_c(y, "_G1")),
    contrastdf |> dplyr::mutate(x = str_c(x, "_G2M"), y = str_c(y, "_G2M"))
  ) |>
    purrr::pmap(\(x, y) {
      res <- ws |>
        dplyr::filter(!is.na(!!sym(sprintf("waldstat_%s_minus_%s", x, y)))) |>
        dplyr::arrange(dplyr::desc(
          !!sym(sprintf("waldstat_%s_minus_%s", x, y))
        ))
      indexbp = limma::ids2indices(phk, res$gene) # order same as res
      res_camerapr <- limma::cameraPR(
        res |> dplyr::pull(!!sym(sprintf("waldstat_%s_minus_%s", x, y))),
        indexbp
      )

      res_camerapr <- res_camerapr |>
        as.data.frame() |>
        tibble::rownames_to_column("gene_set") |>
        tibble::column_to_rownames("gene_set")

      saveRDS(
        res_camerapr,
        file.path(
          ddir,
          sprintf(
            "cameraPR-by-quasipoisson-glm-waldstat-%s-minus-%s_%s.rds",
            x,
            y,
            prefix
          )
        )
      )
      return(res_camerapr)
    })
}

2.5.11 Calculate cameraPR gene set testing across all clusters (1 vs. rest) per sample.

#' Run limma::cameraPR gene set testing for Hallmark gene sets for contrasts of
#' (one vs. rest) on gene lists ranked using quasipoisson Wald statistic between kmeans
#' clusters within a sample using the following criteria:
#'
#' for a set of (one vs. rest) comparisons, if the QPglm Wald statistics:
#' are all positive, use the minimum delta as a marker ranking,
#' are all negative, use the maximum delta as a marker ranking,
#' are mixed positive and negative, use zero as a marker ranking (noninformative), and
#' is NA (phi, rates were zero), we set this to zero, as this indicates no 
#' difference between the clusters (zero is called genewide in this case)
#'
#' @param infile_rds Data frame of Wald statistics comparing clusters, cluster 
#' specific expression rates for a single sample, and sample-specific 
#' dispersion estimates.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
#'
#' @returns List of cameraPR outputs for each tested cluster.
run_cameraPR_with_delta_1vrest_contrasts_across_clusters_within_sample_quasipoisson_glm_simpleaf <- function(
  infile_waldstats,
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  prefix <- gsub(".rds", "", basename(infile_waldstats))
  k = as.numeric(gsub(
    "^waldstat-ranked-genes-across-cluster_within-sample_genewise-rates-within_kmeans([0-9]+).*",
    "\\1",
    prefix
  ))
  prefix <- gsub(
    "waldstat-ranked-genes-across-cluster_within-sample_genewise-rates-within_",
    "",
    prefix
  ) # shorten the filename

  ws <- readRDS(infile_waldstats)
  hk <- msigdbr::msigdbr(species = "human", category = "H") # Hallmark gene collection
  phk = split(x = hk$human_gene_symbol, f = hk$gs_name)

  res_list <- 1:k |>
    purrr::map(\(i) {
      res <- ws |>
        dplyr::mutate(
          !!sym(sprintf(
            "minmaxzero_delta_waldstat_cluster%s_minus_other",
            i
          )) := dplyr::case_when(
            MatrixGenerics::rowMins(as.matrix(dplyr::pick(dplyr::starts_with(sprintf(
              "waldstat_cluster%s_minus",
              i
            ))))) >
              0 ~ MatrixGenerics::rowMins(as.matrix(dplyr::pick(dplyr::starts_with(sprintf(
              "waldstat_cluster%s_minus",
              i
            ))))),
            MatrixGenerics::rowMaxs(as.matrix(dplyr::pick(dplyr::starts_with(sprintf(
              "waldstat_cluster%s_minus",
              i
            ))))) <
              0 ~ MatrixGenerics::rowMaxs(as.matrix(dplyr::pick(dplyr::starts_with(sprintf(
              "waldstat_cluster%s_minus",
              i
            ))))),
            .default = 0
          )
        ) |>
        dplyr::filter(
          !is.na(
            !!sym(sprintf("minmaxzero_delta_waldstat_cluster%i_minus_other", i))
          )
        ) |>
        dplyr::arrange(dplyr::desc(
          !!sym(sprintf("minmaxzero_delta_waldstat_cluster%i_minus_other", i))
        ))

      # Check that we calculated this delta correctly
      stopifnot(all(
        res |>
          dplyr::select(
            dplyr::starts_with(sprintf("waldstat_cluster%s_minus", i)),
            !!sym(sprintf("minmaxzero_delta_waldstat_cluster%s_minus_other", i))
          ) |>
          dplyr::filter(
            !!sym(sprintf(
              "minmaxzero_delta_waldstat_cluster%s_minus_other",
              i
            )) <
              0
          ) <
          0
      ))
      stopifnot(all(
        res |>
          dplyr::select(
            dplyr::starts_with(sprintf("waldstat_cluster%s_minus", i)),
            !!sym(sprintf("minmaxzero_delta_waldstat_cluster%s_minus_other", i))
          ) |>
          dplyr::filter(
            !!sym(sprintf(
              "minmaxzero_delta_waldstat_cluster%s_minus_other",
              i
            )) >
              0
          ) >
          0
      ))

      indexbp = limma::ids2indices(phk, res$gene) # order same as res
      res_camerapr <- limma::cameraPR(
        res |>
          dplyr::pull(
            !!sym(sprintf("minmaxzero_delta_waldstat_cluster%s_minus_other", i))
          ),
        indexbp
      )
      res_camerapr$N_cells <- res |>
        dplyr::pull(!!sym(sprintf("N_cells_%i", i))) |>
        unique() # For nice labels when plotting in heatmap

      res_camerapr <- res_camerapr |>
        as.data.frame() |>
        tibble::rownames_to_column("gene_set")

      saveRDS(
        res_camerapr,
        file.path(
          ddir,
          sprintf(
            "cameraPR-by-QPglm-minmaxzero-delta-waldstat-cluster%i-minus-other_%s.rds",
            i,
            prefix
          )
        )
      )
      saveRDS(
        res,
        file.path(
          ddir,
          sprintf(
            "QPglm-minmaxzero-delta-waldstat-cluster%i-minus-other_%s.rds",
            i,
            prefix
          )
        )
      )
      return(res_camerapr)
    })
  return(res_list)
}

      sid_name = sprintf("Cluster %s (N=%i)", gsub("cluster", "", sid), N_cells)
    ) |>
    ggplot2::ggplot(aes(
      x = sid_name,
      y = gs_name,

      label = NGenes,
      size = FDR
    )) +
    ggplot2::geom_point() +
    ggplot2::theme_bw() +
    ggplot2::theme(
      axis.text.x = ggplot2::element_text(angle = 90, vjust = 0.5, hjust = 1)

    ggplot2::scale_color_manual(values = c(col_high, col_low), drop = FALSE) +
    ggplot2::scale_size(trans = "reverse", range = c(.1, 3)) +
    ggplot2::guides(
      colour = ggplot2::guide_legend(order = 1),

    ) +
    ggplot2::labs(
      x = NULL,
      y = NULL,
      title = title,
      subtitle = sprintf("%s, %s%% FDR per column", "condition", thresh * 100)
    ))

    file.path(
      odir,

    file.path(
      odir,

        "cameraPR-by-QPglm-heatmap_%.2f_%s_%s.png",
        thresh,
        samplename,
        clustername
      )
    ),
    plt,
    width = width,
    height = height
  )

  return(plt)
}

2.6.1 Create gene order table from GTF used for alignment

#' Convert a gtf to a format compatible with Broad's infercnv.
#'
#' Takes the min and max coordinates of each gene in the gtf that was used to
#' run nf-core scrnaseq simpleaf.
#'
#' Removes genes that occur in GTF across multiple chromosomes.
#'
#' @param infile_gtf Path to GTF file used to align scRNAseq samples.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
make_gene_order_file_simpleaf <- function(
  infile_gtf = "/rafalab/drusso/genomes/refgenie_0.12.1_refgenconf_0.12.2_hg38_tmp/gencode_gtf/hg38.gtf.gz",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  d <- data.table::fread(
    infile_gtf,
    skip = 5,
    col.names = c(
      "seqname",
      "source",
      "feature",
      "start",
      "end ",
      "score",
      "strand",
      "frame",
      "attribute"
    )
  )
  out <- d |>
    dplyr::mutate(
      gid = gsub("\"", "", gsub(".*gene_name\ ([^;]*).*", "\\1", attribute))
    ) |> # get gene name
    dplyr::rename(end = `end `) |>
    dplyr::select(gid, seqname, start, end) |>
    dplyr::group_by(gid, seqname) |>
    dplyr::mutate(start = min(start), end = max(end)) |> # span entire gene start to end
    dplyr::ungroup() |>
    dplyr::select(gid, seqname, start, end) |>
    dplyr::distinct()

  # A few genes appear on multiple chromosomes -- filter these out
  multi_genes <- out |>
    dplyr::count(gid) |>
    dplyr::filter(n > 1) |>
    dplyr::pull(gid)
  out <- out |>
    dplyr::filter(!(gid %in% multi_genes))

  readr::write_tsv(out, file.path(ddir, sprintf("hg38_genes_simpleaf.txt")))
  return(out)
}

#' Run Broad's infercnv on scRNAseq samples using the Parental sample as the
#' reference.
#'
#' May need to set options(bitmapType='cairo') to get plots to render,
#' plotting moved to plot_infercnv_simpleaf() function below.
#'
#' @param infile_seurat_rds Path to rds of Seurat object of all samples.
#' @param ddir Processed data directory.
#' @param odir Processed output directory.
run_infercnv_simpleaf <- function(
  infile_seurat_rds = "data/processed/02-scrnaseq/simpleaf-quant-dds-allsamples_filterqc.rds",
  ddir = "data/processed/02-scrnaseq",
  odir = "data/processed/02-scrnaseq"
) {
  prefix <- gsub(".rds", "", basename(infile_seurat_rds))
  dds <- readRDS(infile_seurat_rds)
  counts <- LayerData(dds, assay = "RNA", layer = "counts")

  meta <- dds@meta.data

  # Gene order txt file created above.
  d <- readr::read_tsv("data/processed/02-scrnaseq/hg38_genes_simpleaf.txt")
  gcts <- d$seqname |> table() |> sort()
  names(gcts[gcts < 10]) # scaffold transcripts

  d <- d |>
    filter(!(seqname %in% c(names(gcts[gcts < 10])))) |>
    mutate(
      seqname = factor(seqname, levels = str_c("chr", c(1:23, "X", "Y", "M")))
    ) |>
    arrange(seqname, start)

  infercnv_obj = CreateInfercnvObject(
    raw_counts_matrix = counts,
    annotations_file = (meta |> select(orig.ident)),
    delim = "\t",
    ref_group_names = c("Parental"),
    gene_order_file = (d |> tibble::column_to_rownames("gid")),
    chr_exclude = c(names(gcts[gcts < 10]))
  )

  out_dir = tempfile()
  infercnv_obj_default = infercnv::run(
    infercnv_obj,
    cutoff = 0.1,
    out_dir = ddir,
    cluster_by_groups = TRUE,
    plot_steps = FALSE,
    denoise = TRUE,
    HMM = FALSE,
    no_plot = TRUE, # can be set to FALSE to generate plot in this step
    no_prelim_plot = TRUE,
    png_res = 60
  )
}
